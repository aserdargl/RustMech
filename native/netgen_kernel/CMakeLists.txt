cmake_minimum_required(VERSION 3.20)
project(RustMechNetgen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(zlib EXCLUDE_FROM_ALL)
set(ZLIB_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/zlib;${CMAKE_CURRENT_BINARY_DIR}/zlib")
set(ZLIB_LIBRARIES zlibstatic)
set(USE_SUPERBUILD OFF CACHE BOOL "" FORCE)
set(USE_GUI OFF CACHE BOOL "" FORCE)
set(USE_OCC OFF CACHE BOOL "" FORCE)
set(USE_PYTHON OFF CACHE BOOL "" FORCE)
set(USE_MPI OFF CACHE BOOL "" FORCE)
set(USE_NATIVE_ARCH OFF CACHE BOOL "" FORCE)
set(USE_CCACHE OFF CACHE BOOL "" FORCE)
set(BUILD_STUB_FILES OFF CACHE BOOL "" FORCE)
set(NG_FP_CONTRACT OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream)
set_target_properties(nglib PROPERTIES OUTPUT_NAME rustmech_netgen)
set_target_properties(ngcore PROPERTIES OUTPUT_NAME rustmech_ngcore)
target_sources(nglib PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/kernel_api.cpp")
target_include_directories(nglib PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(nglib ngcore PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
add_executable(point_tree_tests EXCLUDE_FROM_ALL point_tree_tests.cpp)
target_include_directories(point_tree_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/upstream/libsrc" "${CMAKE_CURRENT_SOURCE_DIR}/upstream/libsrc/include" "${CMAKE_CURRENT_BINARY_DIR}/upstream")
target_link_libraries(point_tree_tests PRIVATE nglib ngcore)
set_target_properties(point_tree_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
